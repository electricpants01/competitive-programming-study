gcd_lcm.cpp 
#include <iostream>
using namespace std;
typedef long long ll;

// MCD con el Algoritmo de Euclides (recursivo)
ll mcd(ll a, ll b) {
    return b == 0 ? a : mcd(b, a % b);
}

// MCD iterativo (equivalente, sin recursión)
ll mcd_iter(ll a, ll b) {
    while (b != 0) {
        ll r = a % b;
        a = b; b = r;
    }
    return a;
}

// MCM usando la fórmula mcd × mcm = a × b
ll mcm(ll a, ll b) {
    return a / mcd(a, b) * b; // divide primero para evitar overflow
}

int main() {
    ll a = 252, b = 198;
    cout << "MCD(" << a << "," << b << ") = " << mcd(a, b) << endl;
    cout << "MCM(" << a << "," << b << ") = " << mcm(a, b) << endl;
    return 0;
}
🔁
Recursivo vs Iterativo: Ambos son O(log n). El recursivo es más elegante, el iterativo evita el stack overflow para recursión profunda.
⚠️
Divide primero: a/mcd(a,b)*b evita overflow que produciría a*b/mcd(a,b).
⚡
C++17 built-in: #include <numeric> y usar __gcd(a,b) o std::gcd(a,b).
