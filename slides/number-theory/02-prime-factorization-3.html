#include <iostream>
#include <vector>
#include <map>
using namespace std;

// Retorna los factores primos de n
vector<pair<int,int>> factorizar(int n) {
    vector<pair<int,int>> factores;
    for (int d = 2; d * d <= n; d++) {
        if (n % d == 0) {
            int exp = 0;
            while (n % d == 0) { // contar cuántas veces divide
                exp++;
                n /= d;
            }
            factores.push_back({d, exp}); // guardar (primo, exponente)
        }
    }
    if (n > 1) factores.push_back({n, 1}); // factor primo restante
    return factores;
}

int main() {
    int n = 360;
    auto f = factorizar(n);
    cout << n << " = ";
    for (auto [p, e] : f)
        cout << p << "^" << e << " * ";
    return 0;
}
