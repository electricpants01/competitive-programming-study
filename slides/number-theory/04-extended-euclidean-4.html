#include <iostream>
using namespace std;
typedef long long ll;

// Retorna mcd(a,b) y encuentra x,y tal que a*x + b*y = mcd(a,b)
ll mcd_ext(ll a, ll b, ll &x, ll &y) {
    if (b == 0) {
        x = 1; y = 0; // Caso base: a*1 + 0*0 = a
        return a;
    }
    ll x1, y1;
    ll g = mcd_ext(b, a % b, x1, y1); // Llamada recursiva
    x = y1;                             // Actualizar x
    y = x1 - (a / b) * y1;             // Actualizar y
    return g;
}

// Calcula el inverso modular de a módulo m
ll inv_mod(ll a, ll m) {
    ll x, y;
    ll g = mcd_ext(a, m, x, y);
    if (g != 1) return -1; // No existe inverso
    return (x % m + m) % m; // Normalizar a [0, m-1]
}

int main() {
    ll a = 35, b = 15, x, y;
    ll g = mcd_ext(a, b, x, y);
    cout << "mcd = " << g << ", x = " << x << ", y = " << y << endl;

    cout << "inv(3, 7) = " << inv_mod(3, 7) << endl;
    return 0;
}
Caso base: Cuando b=0, mcd=a, x=1, y=0 (trivialmente a·1 + 0·0 = a)
Recurrencia: Los coeficientes se calculan a partir de la llamada anterior: x = y1, y = x1 − q·y1
Normalizar: (x%m+m)%m convierte un x negativo a su equivalente positivo
$ ./euclides_extendido
mcd = 5, x = 1, y = -2
inv(3, 7) = 5
→ 35·1 + 15·(−2) = 5 ✓
→ 3·5 = 15 ≡ 1 (mod 7) ✓
