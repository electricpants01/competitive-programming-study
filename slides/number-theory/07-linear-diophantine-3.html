#include <iostream>
using namespace std;
typedef long long ll;

ll mcd_ext(ll a, ll b, ll &x, ll &y) {
    if (!b) { x=1; y=0; return a; }
    ll x1, y1;
    ll g = mcd_ext(b, a%b, x1, y1);
    x = y1; y = x1 - (a/b)*y1;
    return g;
}

// Resuelve ax + by = c
// Retorna false si no tiene solución
bool diofantica(ll a, ll b, ll c,
                   ll &x, ll &y, ll &dx, ll &dy) {
    ll g = mcd_ext(a, b, x, y);
    if (c % g != 0) return false; // sin solución
    ll k = c / g;
    x *= k;       // solución particular
    y *= k;
    dx = b / g;   // paso para x (familia completa)
    dy = a / g;   // paso para y
    return true;
}

int main() {
    ll a=6, b=9, c=12;
    ll x, y, dx, dy;
    if (diofantica(a, b, c, x, y, dx, dy)) {
        cout << "Solución: x=" << x << " y=" << y << endl;
        cout << "Familia: x(t)=" << x << "+" << dx << "t" << endl;
        cout << "         y(t)=" << y << "-" << dy << "t" << endl;
    } else cout << "Sin solución entera" << endl;
}
Retorna dx, dy: los pasos de la familia de soluciones. x(t) = x + dx·t, y(t) = y − dy·t
c % g != 0: condición necesaria. Si el MCD no divide a c, no existe ningún (x,y) entero que satisfaga la ecuación.

📌 Resumen
1. mcd_ext → solución base
2. Verificar c % mcd == 0
3. Escalar por c/mcd
4. Familia: ±b/mcd, ∓a/mcd
Complejidad: O(log min(a,b))
